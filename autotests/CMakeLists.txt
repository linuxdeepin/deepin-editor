# autotests/CMakeLists.txt — Google Test suite for deepin-editor.
# Standalone CMake project (built via autotests/run-ut.sh: `cmake <autotests>`).
# deepin-editor has no shared library target, so each test executable compiles
# the application sources directly (same proven approach as the existing
# tests/ target), plus the per-class Google Test files under autotests/<module>/.

cmake_minimum_required(VERSION 3.16)
project(deepin_editor_autotests VERSION 1.0.0 LANGUAGES CXX C)

set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(UnitTestUtils)

enable_testing()

# Project root (parent of autotests/)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)

# ---- Compile definitions normally provided by the root CMakeLists.txt ----
# deepin-editor sources reference KF5_HIGHLIGHT_PATH and LINGLONG_PREFIX.
if(NOT DEFINED CMAKE_INSTALL_PREFIX OR CMAKE_INSTALL_PREFIX STREQUAL "")
    set(CMAKE_INSTALL_PREFIX "/usr/local")
endif()
add_definitions(-DLINGLONG_PREFIX=\"${CMAKE_INSTALL_PREFIX}/\")
set(KF5_HIGHLIGHT_INSTALL_PATH "${CMAKE_INSTALL_PREFIX}/share/deepin-editor/org.kde.syntax-highlighing")
add_definitions(-DKF5_HIGHLIGHT_PATH=\"${KF5_HIGHLIGHT_INSTALL_PATH}\")

# ---- Version header (environments.h) generated by the root project; regenerate
# here so the standalone build can compile sources that #include <environments.h>.
set(VERSION "6.0.0" CACHE STRING "Application version")
configure_file("${PROJECT_ROOT}/src/environments.h.in"
               "${CMAKE_BINARY_DIR}/environments.h" @ONLY)
include_directories(${CMAKE_BINARY_DIR})

# ---- Find packages (mirrors tests/CMakeLists.txt) ----
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
message("   >>> [autotests] Found Qt version: ${QT_VERSION_MAJOR}")
set(QT_DESIRED_VERSION ${QT_VERSION_MAJOR})
if(QT_VERSION_MAJOR MATCHES 6)
    set(DTK_VERSION_MAJOR 6)
    set(KF_VERSION_MAJOR 6)
else()
    set(DTK_VERSION_MAJOR "")
    set(KF_VERSION_MAJOR 5)
endif()

find_package(PkgConfig REQUIRED)

set(qt_required_components Gui Widgets DBus Concurrent PrintSupport Svg Xml Test)
if(QT_DESIRED_VERSION MATCHES 6)
    list(APPEND qt_required_components Core5Compat)
endif()
find_package(Qt${QT_DESIRED_VERSION} REQUIRED COMPONENTS ${qt_required_components})
find_package(Dtk${DTK_VERSION_MAJOR} REQUIRED COMPONENTS Widget Core)
find_package(KF${KF_VERSION_MAJOR}Codecs REQUIRED)
find_package(KF${KF_VERSION_MAJOR}SyntaxHighlighting REQUIRED)
find_package(GTest REQUIRED)
include(GoogleTest)
find_package(ICU COMPONENTS i18n uc REQUIRED)

pkg_check_modules(chardet REQUIRED chardet)
include_directories(${chardet_INCLUDE_DIRS})
link_directories(${chardet_LIBRARY_DIRS})

# Link + runtime libs (uchardet/chardet runtimes live in system lib dir)
link_libraries(uchardet)
link_libraries(chardet)

set(LINK_LIBS
    Qt${QT_VERSION_MAJOR}::Gui
    Qt${QT_VERSION_MAJOR}::Widgets
    Qt${QT_VERSION_MAJOR}::Xml
    Qt${QT_VERSION_MAJOR}::Svg
    Qt${QT_VERSION_MAJOR}::Concurrent
    Qt${QT_VERSION_MAJOR}::PrintSupport
    Qt${QT_VERSION_MAJOR}::DBus
    Qt${QT_VERSION_MAJOR}::Test
    Dtk${DTK_VERSION_MAJOR}::Widget
    Dtk${DTK_VERSION_MAJOR}::Core
    KF${KF_VERSION_MAJOR}::Codecs
    KF${KF_VERSION_MAJOR}::SyntaxHighlighting
)
if(QT_DESIRED_VERSION MATCHES 6)
    list(APPEND LINK_LIBS Qt${QT_DESIRED_VERSION}::Core5Compat)
endif()

# ---- Coverage + access flags (Debug only) ----
# lcov capture (run-ut.sh step 5) and report_generator depend on gcov instrumentation.
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -g -fprofile-arcs -ftest-coverage -fno-inline -O0")
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -g -fprofile-arcs -ftest-coverage -O0")
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -lgcov")
    message(STATUS "[autotests] Code coverage enabled (gcov instrumentation)")
endif()
# Allow tests to touch private/protected members (matches existing tests/ target).
ADD_COMPILE_OPTIONS(-fno-access-control)

# ---- Source layout ----
include_directories(${PROJECT_ROOT}/src)
include_directories(${PROJECT_ROOT}/src/encodes)
include_directories(${PROJECT_ROOT}/src/common)
include_directories(${PROJECT_ROOT}/src/controls)
include_directories(${PROJECT_ROOT}/src/editor)
include_directories(${PROJECT_ROOT}/src/thememodule)
include_directories(${PROJECT_ROOT}/src/widgets)
include_directories(${PROJECT_ROOT}/src/basepub)
include_directories(${PROJECT_ROOT}/src/dbus)
include_directories(${GTEST_INCLUDE_DIRS})

# Application sources: everything under src/ except the two main() entry points
# and encoding.cpp (declared in encodes/ and excluded to avoid duplicate symbols).
file(GLOB_RECURSE APP_SRC  "${PROJECT_ROOT}/src/*.cpp" "${PROJECT_ROOT}/src/basepub/*.c")
file(GLOB_RECURSE APP_HDRS "${PROJECT_ROOT}/src/*.h")
list(REMOVE_ITEM APP_SRC
    "${PROJECT_ROOT}/src/main.cpp"
    "${PROJECT_ROOT}/src/dedit/main.cpp"
    "${PROJECT_ROOT}/src/common/encoding.cpp")

# Per-class Google Test files (autotests/<module>/test_*.cpp)
file(GLOB_RECURSE TEST_SRC "${CMAKE_CURRENT_SOURCE_DIR}/test_*.cpp")

set(APP_QRC "${PROJECT_ROOT}/src/deepin-editor.qrc")
set(STUB_SHADOW "${CMAKE_CURRENT_SOURCE_DIR}/3rdparty/stub/stub-shadow.cpp")

add_executable(deepin-editor-autotest
    ${APP_HDRS}
    ${APP_SRC}
    ${TEST_SRC}
    ${APP_QRC}
    ${STUB_SHADOW}
)

target_link_libraries(deepin-editor-autotest
    PRIVATE
        gmock gtest
    ${LINK_LIBS}
    ${GTEST_LIBRARIES}
    ICU::i18n
    ICU::uc
    -lpthread
    -lm
    dl
    uchardet
    chardet
)

target_include_directories(deepin-editor-autotest PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/3rdparty/stub"
)

# Per-test isolation + headless Qt.
gtest_discover_tests(deepin-editor-autotest
    PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_LOGGING_RULES=*=false"
)

message(STATUS "[autotests] Configured with ${TEST_SOURCES_COUNT} test sources")
